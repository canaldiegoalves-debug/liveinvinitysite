function requestIp(request) {
  return String(
    request.headers["x-forwarded-for"] ||
    request.socket?.remoteAddress ||
    "unknown"
  ).split(",")[0].trim();
}

  return false;

function registerAdminLoginFailure(request) {
  const ip = requestIp(request);
  const now = Date.now();
  const entry = adminLoginAttempts.get(ip);

  if (!entry || now - entry.startedAt > ADMIN_LOGIN_WINDOW_MS) {
    adminLoginAttempts.set(ip, { count: 1, startedAt: now });
    return;
  }

  entry.count += 1;
  adminLoginAttempts.set(ip, entry);
}

function clearAdminLoginFailures(request) {
  adminLoginAttempts.delete(requestIp(request));
}

function nestedValue(object, paths) {
  for (const candidate of paths) {
    const parts = candidate.split(".");
    let value = object;

    for (const part of parts) {
      value = value?.[part];
    }

    if (value !== undefined && value !== null && value !== "") {
      return value;
    }
  }

  return null;
}

function findPlanInPayload(body) {
  const serialized = JSON.stringify(body);

  // 1. Procura por IDs de checkout/oferta conhecidos no payload
  for (const [code, plan] of Object.entries(CAKTO_CHECKOUT_MAP)) {
    if (code && serialized.includes(code)) return plan;
  }

  // 2. Procura pelo valor pago (Amount / Price)
  const eventData = caktoEventData(body);
  const price = eventData.amountPaid || Number(body.amount || body.price || body.data?.amount || body.data?.price || 0);

  if (price > 0) {
    if (price <= 70 || serialized.includes("67")) return "basic";
    if (price > 70 && price <= 120 || serialized.includes("97")) return "pro";
    if (price > 120 || serialized.includes("147")) return "premium";
  }

  // 3. Procura por palavras-chave de nome de produto/oferta
  const lowerSer = serialized.toLowerCase();
  if (lowerSer.includes("básico") || lowerSer.includes("basico") || lowerSer.includes("starter") || lowerSer.includes("67")) return "basic";
  if (lowerSer.includes("premium") || lowerSer.includes("infinity") || lowerSer.includes("vip") || lowerSer.includes("147")) return "premium";
  if (lowerSer.includes("pro") || lowerSer.includes("97")) return "pro";

  // 4. Fallback automático para pagamentos ou assinaturas ativas da Cakto
  const status = String(eventData.paymentStatus || "").toLowerCase();
  const type = String(eventData.eventType || "").toLowerCase();
  if (status.includes("paid") || status.includes("approved") || type.includes("purchase_approved") || type.includes("subscription_created")) {
    return "pro";
  }

  return null;
}

function caktoEventData(body) {
  const eventType = String(nestedValue(body, [
    "event",
    "event_type",
    "type",
    "custom_id",
    "data.event",
    "data.custom_id"
  ]) || "").toLowerCase();

  const offerId = String(nestedValue(body, [
    "offer.id",
    "offer_id",
    "data.offer.id",
    "data.offer_id",
    "order.offer.id",
    "data.order.offer.id"
  ]) || "");

  const orderId = String(nestedValue(body, [
    "order.id",
    "order_id",
    "data.order.id",
    "data.id",
    "id"
  ]) || "");

  const email = normalizeEmail(nestedValue(body, [
    "customer.email",
    "buyer.email",
    "order.customer.email",
    "data.customer.email",
    "data.buyer.email",
    "data.order.customer.email",
    "email"
  ]));

  const paymentStatus = String(nestedValue(body, [
    "order.status",
    "status",
    "data.order.status",
    "data.status"
  ]) || "").toLowerCase();

  const eventId = String(nestedValue(body, [
    "event_id",
    "event.id",
    "data.event_id",
    "data.event.id"
  ]) || `${orderId || "no-order"}:${eventType || paymentStatus || "unknown"}`);

  // Extrai valor bruto pago pelo cliente
  const amountPaidVal = nestedValue(body, [
    "order.amount",
    "order.total",
    "amount",
    "total_value",
    "data.order.amount"
  ]);
  let amountPaid = amountPaidVal ? Number(amountPaidVal) : null;
  if (amountPaid && amountPaid > 1000 && !String(amountPaidVal).includes(".")) {
    amountPaid = amountPaid / 100;
  }

  // Extrai lucro real recebido pelo vendedor
  const netAmountVal = nestedValue(body, [
    "order.net_value",
    "order.net_amount",
    "order.commission",
    "net_value",
    "net_amount",
    "commission",
    "commission_value",
    "comissao",
    "comissao_valor",
    "data.order.net_value",
    "data.order.commission"
  ]);
  let netAmount = netAmountVal ? Number(netAmountVal) : null;
  if (netAmount && netAmount > 1000 && !String(netAmountVal).includes(".")) {
    netAmount = netAmount / 100;
  }

  const rawPhone = String(nestedValue(body, [
    "customer.phone",
    "buyer.phone",
    "order.customer.phone",
    "data.customer.phone",
    "data.buyer.phone",
    "data.order.customer.phone",
    "phone"
  ]) || "");
  const phone = rawPhone.replace(/\D/g, "");

  return { eventId, eventType, offerId, orderId, email, phone, paymentStatus, amountPaid, netAmount };
}

function sanitizeVersion(value) {
  const version = String(value || "").trim().replace(/^v/i, "");
  return /^\d+\.\d+\.\d+(?:-[a-z0-9.-]+)?$/i.test(version)
    ? version
    : null;
}

function compareVersions(left, right) {
  const a = String(left || "0.0.0").split("-")[0].split(".").map(Number);
  const b = String(right || "0.0.0").split("-")[0].split(".").map(Number);

  for (let index = 0; index < 3; index += 1) {
    const difference = (a[index] || 0) - (b[index] || 0);
    if (difference !== 0) return difference;
  }

  return 0;
}

function readBody(request) {
  return new Promise((resolve, reject) => {
    let data = "";

    request.on("data", chunk => {
      data += chunk;
      if (data.length > 50_000_000) { // 50MB
        reject(new Error("Payload muito grande."));
        request.destroy();
      }
    });

    request.on("end", () => {
      if (!data) return resolve({});
      try {
        resolve(JSON.parse(data));
      } catch {
        try {
          const params = new URLSearchParams(data);
          const obj = {};
          for (const [k, v] of params.entries()) obj[k] = v;
          if (Object.keys(obj).length > 0) return resolve(obj);
        } catch (e2) {}
        reject(new Error("JSON inválido."));
      }
    });

    request.on("error", reject);
  });
}

function safeEqual(left, right) {
  const a = Buffer.from(String(left));
  const b = Buffer.from(String(right));
  return a.length === b.length && crypto.timingSafeEqual(a, b);
}

function rowToLicense(row) {
  const expiresAt = row.expires_at ? new Date(row.expires_at) : null;
  const remainingDays = expiresAt
    ? Math.max(0, Math.ceil((expiresAt.getTime() - Date.now()) / 86_400_000))
    : Number(row.duration_days);

  let status = row.status;
  if (!row.active || status === "revoked") status = "revoked";
  else if (expiresAt && expiresAt.getTime() <= Date.now()) status = "expired";
  else if (row.activated_at) status = "active";
  else status = "pending";

  const isLiveCam = String(row.license_key || "").startsWith("LIVECAM");

  return {
    id: row.id,
    email: row.access_email || row.email,
    accountEmail: row.account_email || row.email,
    accessEmail: row.access_email || row.email,
    key: row.license_key,
    product: isLiveCam ? "Live Cam" : "Live Infinity",
    productCode: isLiveCam ? "livecam" : "liveinfinity",
    plan: row.plan,
    durationDays: Number(row.duration_days),
    active: Boolean(row.active),
    status,
    createdAt: toIso(row.created_at),
    activatedAt: toIso(row.activated_at),
    expiresAt: toIso(row.expires_at),
    remainingDays,
    deviceId: row.device_id || null,
    note: row.note || "",
    phone: row.phone || "",
    amountPaid: Number(row.amount_paid || 0),
    netAmount: Number(row.net_amount || 0),
    totalGmv: Number(row.total_gmv || 0),
    paymentMethod: row.payment_method || "",
    lastValidationAt: toIso(row.last_validation_at),
    planName: planDefinition(row.plan).name,
    monthlyPrice: planDefinition(row.plan).monthlyPrice,
    keyLimit: planDefinition(row.plan).keyLimit
  };
}

function validateLicenseRecord(row, email, key, deviceId, deviceFingerprint) {
  if (!row) return { ok: false, error: "Chave não encontrada." };
  if (!row.active || row.status === "revoked") {
    return { ok: false, error: "Esta licença foi bloqueada." };
  }
  const inputEmail = normalizeEmail(email);
  const rowAccess = normalizeEmail(row.access_email);
  const rowAccount = normalizeEmail(row.account_email);
  const rowEmail = normalizeEmail(row.email);

  if (inputEmail && inputEmail !== rowAccess && inputEmail !== rowAccount && inputEmail !== rowEmail) {
    return { ok: false, error: "O e-mail não corresponde à chave." };
  }
  if (row.license_key !== String(key || "").trim().toUpperCase()) {
    return { ok: false, error: "Chave inválida." };
  }
  if (row.device_id && deviceId && row.device_id !== deviceId) {
    const sameComputer =
      row.device_fingerprint &&
      deviceFingerprint &&
      safeEqual(row.device_fingerprint, deviceFingerprint);

    if (!sameComputer) {
      return {
        ok: false,
        error: row.device_fingerprint
          ? "Esta chave já está vinculada a outro computador."
          : "Ativação antiga detectada. Libere o PC uma única vez no painel administrativo e tente novamente.",
        legacyDeviceBinding: !row.device_fingerprint
      };
    }
  }
  if (row.expires_at && new Date(row.expires_at).getTime() <= Date.now()) {
    return { ok: false, error: "A licença expirou.", expired: true };
  }
  return { ok: true };
}

function serveAdminStatic(requestPath, response) {
  if (requestPath === ADMIN_PANEL_PATH) {
    response.writeHead(302, {
      Location: `${ADMIN_PANEL_PATH}/`,
      "Cache-Control": "no-store"
    });
    response.end();
    return;
  }

  let relativePath = requestPath;
  if (relativePath.startsWith(ADMIN_PANEL_PATH)) {
    relativePath = relativePath.slice(ADMIN_PANEL_PATH.length);
  }
  if (!relativePath || relativePath === "/") {
    relativePath = "/index.html";
  }

  const safe = path
    .normalize(relativePath)
    .replace(/^(\.\.[/\\])+/, "");

  let filePath = path.join(PUBLIC_DIR, safe);

  if (fs.existsSync(filePath) && fs.statSync(filePath).isDirectory()) {
    filePath = path.join(filePath, "index.html");
  }

  if (!filePath.startsWith(PUBLIC_DIR) || !fs.existsSync(filePath)) {
    text(response, 404, "Página não encontrada.");
    return;
  }

  const extension = path.extname(filePath).toLowerCase();
  const types = {
    ".html": "text/html; charset=utf-8",
    ".css": "text/css; charset=utf-8",
    ".js": "application/javascript; charset=utf-8",
    ".png": "image/png"
  };

  response.setHeader(
    "Content-Security-Policy",
    "default-src 'self'; img-src 'self' data:; style-src 'self'; script-src 'self'; connect-src 'self'; frame-ancestors 'none';"
  );
  response.setHeader("X-Frame-Options", "DENY");
  response.setHeader("X-Content-Type-Options", "nosniff");
  response.setHeader("Referrer-Policy", "no-referrer");
  response.setHeader("Cache-Control", "no-store");

  text(
    response,
    200,
    fs.readFileSync(filePath),
    types[extension] || "application/octet-stream"
  );
}

    CREATE TABLE IF NOT EXISTS live_sessions (
      session_id VARCHAR(255) PRIMARY KEY,
      license_id INT NOT NULL,
      streamer_name VARCHAR(255) NULL,
      platform VARCHAR(100) NULL,
      viewers INT NOT NULL DEFAULT 0,
      vendas INT NOT NULL DEFAULT 0,
      gmv DECIMAL(15,2) NOT NULL DEFAULT 0.00,
      created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
      updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
      INDEX idx_ls_license (license_id),
      INDEX idx_ls_updated (updated_at)
    ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
  `);
}

async function ensureSupportMessagesTable() {
  await pool.query(`
    CREATE TABLE IF NOT EXISTS support_messages (
      id VARCHAR(255) PRIMARY KEY,
      ticket_id VARCHAR(255) NOT NULL,
      sender VARCHAR(50) NOT NULL,
      message TEXT NOT NULL,
      created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
      INDEX idx_sm_ticket (ticket_id),
      INDEX idx_sm_created (created_at)
    ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
  `);
}

async function ensurePhoneColumn() {
  try {
    const [columns] = await pool.query(
      "SHOW COLUMNS FROM licenses LIKE 'phone'"
    );
    if (columns.length === 0) {
      console.log("[Banco de Dados] Adicionando coluna 'phone' na tabela licenses...");
      await pool.query(
        "ALTER TABLE licenses ADD COLUMN phone VARCHAR(100) NULL AFTER email"
      );
    }
  } catch (err) {
    console.error("[Banco de Dados] Erro ao garantir coluna 'phone':", err.message);
  }
}

async function reprocessPendingWebhooks() {
  console.log("[Webhook Cakto] Verificando eventos pendentes no banco de dados...");
  try {
    const [allEvents] = await pool.query(
      "SELECT DISTINCT customer_email, event_type, payment_status, processed FROM cakto_webhook_events"
    );
    console.log("[Webhook Debug] Todos os eventos gravados no banco de dados:", allEvents);

    const [rows] = await pool.query(
      "SELECT * FROM cakto_webhook_events WHERE processed = 0 OR processed IS NULL"
    );

    if (rows.length === 0) {
      console.log("[Webhook Cakto] Nenhum evento pendente encontrado.");
      return;
    }

    console.log(`[Webhook Cakto] Encontrados ${rows.length} eventos pendentes para reprocessamento.`);

    for (const row of rows) {
      try {
        const body = typeof row.payload === "string" ? JSON.parse(row.payload) : row.payload;
        const event = caktoEventData(body);
        const plan = row.mapped_plan || findPlanInPayload(body);

        console.log(`[Webhook Reprocessar] Analisando evento: "${row.event_id}" | Email: "${event.email}" | Plano Mapeado: "${plan || 'Não encontrado'}"`);

        if (!event.email || !plan) {
          const errorMsg = `Dados insuficientes: email="${event.email || 'nulo'}", plano="${plan || 'nulo'}"`;
          console.warn(`[Webhook Reprocessar] Ignorando evento "${row.event_id}": ${errorMsg}`);
          await pool.execute(
            "UPDATE cakto_webhook_events SET processed = 0, processing_error = ? WHERE event_id = ?",
            [errorMsg, row.event_id]
          );
          continue;
        }

        const lowerStatus = String(event.paymentStatus || "").trim().toLowerCase();
        const lowerType = String(event.eventType || "").trim().toLowerCase();

        const isPaid = lowerStatus === "paid" ||
                       lowerStatus === "approved" ||
                       lowerStatus === "active" ||
                       lowerStatus === "completed" ||
                       lowerStatus === "success" ||
                       lowerType === "charge.paid" ||
                       lowerType === "purchase_approved" ||
                       lowerType === "subscription_created" ||
                       (lowerType === "subscription.status_changed" && lowerStatus === "active");

        const isRevoked = lowerStatus === "refunded" ||
                           lowerStatus === "chargeback" ||
                           lowerStatus === "cancelled" ||
                           lowerStatus === "expired" ||
                           lowerType === "charge.refunded" ||
                           lowerType === "subscription.canceled";

        let processed = 0;
        let processingError = null;

        if (isPaid) {
          await upsertCustomerAccount(event.email, plan);

          const [existing] = await pool.execute(
            `SELECT * FROM licenses WHERE access_email=?`,
            [event.email]
          );

          if (existing.length > 0) {
            const licenseRow = existing[0];
            const durationDays = 30;
            const newExpires = new Date(Date.now() + durationDays * 86400000);
            const amountPaid = event.amountPaid || (plan === "premium" ? 147.00 : plan === "pro" ? 97.00 : 67.00);
            const netAmount = event.netAmount || (amountPaid * 0.92);

            await pool.execute(
              `UPDATE licenses
               SET expires_at=?,
                   active=1,
                   status='active',
                   plan=?,
                   duration_days=?,
                   amount_paid=?,
                   net_amount=?
               WHERE id=?`,
              [newExpires, plan, durationDays, amountPaid, netAmount, licenseRow.id]
            );

            await upsertCustomerAccount(event.email, plan);
            await pool.execute(
              `UPDATE customer_accounts
               SET subscription_status='active',
                   current_period_end=?
               WHERE email=?`,
              [newExpires, event.email]
            );
            console.log(`[Webhook Reprocessar] Licença existente estendida com SUCESSO para: ${event.email}`);
          } else {
            const id = crypto.randomUUID();
            let key;
            const durationDays = 30;
            const expiresAt = new Date(Date.now() + durationDays * 86400000);
            const amountPaid = event.amountPaid || (plan === "premium" ? 147.00 : plan === "pro" ? 97.00 : 67.00);
            const netAmount = event.netAmount || (amountPaid * 0.92);

            for (let attempt = 0; attempt < 10; attempt += 1) {
              key = generateKey(plan);
              try {
                await pool.execute(
                  `INSERT INTO licenses
                   (id,account_email,access_email,email,phone,license_key,plan,duration_days,note,amount_paid,net_amount,payment_method,expires_at,status,active)
                   VALUES (?,?,?,?,?,?,?,?,?,?,?,?,?,?,?)`,
                  [
                    id,
                    event.email,
                    event.email,
                    event.email,
                    event.phone || null,
                    key,
                    plan,
                    durationDays,
                    `Criado via reprocessamento webhook Cakto: ${event.orderId || ""}`,
                    amountPaid,
                    netAmount,
                    "Cakto",
                    expiresAt,
                    "pending",
                    1
                  ]
                );
                break;
              } catch (error) {
                if (error.code !== "ER_DUP_ENTRY" || attempt === 9) throw error;
              }
            }

            await pool.execute(
              `UPDATE customer_accounts
               SET subscription_status='active',
                   current_period_end=?
               WHERE email=?`,
              [expiresAt, event.email]
            );
            console.log(`[Webhook Reprocessar] Nova licença criada com SUCESSO para: ${event.email} (Key: ${key})`);
          }
          processed = 1;
        } else if (isRevoked) {
          await pool.execute(
            `UPDATE licenses SET active=0, status='revoked' WHERE access_email=?`,
            [event.email]
          );
          await pool.execute(
            `UPDATE customer_accounts SET subscription_status='cancelled' WHERE email=?`,
            [event.email]
          );
          console.log(`[Webhook Reprocessar] Acesso revogado para: ${event.email}`);
          processed = 1;
        } else {
          processingError = `Ignorado: Status/Evento não mapeado (${lowerType} / ${lowerStatus})`;
        }

        await pool.execute(
          `UPDATE cakto_webhook_events
           SET processed=?,
               processing_error=?,
               processed_at=NOW()
           WHERE event_id=?`,
          [processed, processingError, row.event_id]
        );

      } catch (err) {
        console.error(`[Webhook Reprocessar] Erro no evento "${row.event_id}":`, err.message);
        await pool.execute(
          "UPDATE cakto_webhook_events SET processed=0, processing_error=? WHERE event_id=?",
          [`Erro no reprocessamento: ${err.message}`, row.event_id]
        );
      }
    }
  } catch (globalErr) {
    console.error("[Webhook Reprocessar] Erro global ao reprocessar:", globalErr.message);
  }
}

const server = http.createServer(async (request, response) => {
  if (request.method === "OPTIONS") {
    response.writeHead(204, {
      "Access-Control-Allow-Origin": "*",
      "Access-Control-Allow-Headers": "Content-Type, Authorization",
      "Access-Control-Allow-Methods": "GET, POST, PATCH, DELETE, OPTIONS"
    });
    response.end();
    return;
  }

  const url = new URL(request.url, `http://${request.headers.host}`);
  url.pathname = url.pathname.replace(/^\/painel-seguro-liveinfinity/, "") || "/";

  try {

    if (url.pathname === "/downloads/live-infinity.zip") {
      const zipPath = path.join(PUBLIC_DIR, "downloads", "live-infinity.zip");
      if (fs.existsSync(zipPath)) {
        response.writeHead(200, {
          "Content-Type": "application/zip",
          "Content-Disposition": "attachment; filename=live-infinity.zip",
          "Access-Control-Allow-Origin": "*"
        });
        response.end(fs.readFileSync(zipPath));
        return;
      }
    }

    if (url.pathname === "/downloads/livecam.zip") {
      const zipPath = path.join(PUBLIC_DIR, "downloads", "livecam.zip");
      if (fs.existsSync(zipPath)) {
        response.writeHead(200, {
          "Content-Type": "application/zip",
          "Content-Disposition": "attachment; filename=livecam.zip",
          "Access-Control-Allow-Origin": "*"
        });
        response.end(fs.readFileSync(zipPath));
        return;
      }
    }

    if (url.pathname === "/download" || url.pathname === "/downloads") {
      const filePath = path.join(PUBLIC_DIR, "download.html");
      text(response, 200, fs.readFileSync(filePath), "text/html; charset=utf-8");
      return;
    }

    if (url.pathname === "/ativar-liveinfinity" || url.pathname === "/gerar-chave-liveinfinity" || url.pathname === "/ativar") {
      const filePath = path.join(PUBLIC_DIR, "ativar-liveinfinity.html");
      text(response, 200, fs.readFileSync(filePath), "text/html; charset=utf-8");
      return;
    }

    if (url.pathname === "/ativar-livecam" || url.pathname === "/gerar-chave-livecam") {
      const filePath = path.join(PUBLIC_DIR, "ativar-livecam.html");
      text(response, 200, fs.readFileSync(filePath), "text/html; charset=utf-8");
      return;
    }

    if ((url.pathname === "/api/public/claim-license" || url.pathname === "/api/public/claim-livecam" || url.pathname === "/api/public/claim-liveinfinity") && request.method === "POST") {
      try {
        const body = await readBody(request);
        const email = normalizeEmail(body.email);
        
        const referer = String(request.headers.referer || "").toLowerCase();
        const bodyProd = String((body && body.product) || "").toLowerCase();
        const urlProd = String(url.searchParams.get("product") || "").toLowerCase();
        const isLiveCam = url.pathname === "/api/public/claim-livecam" || bodyProd === "livecam" || urlProd === "livecam" || referer.includes("livecam");
        const requestedProduct = isLiveCam ? "livecam" : "liveinfinity";

        if (!email || !email.includes("@")) {
          json(response, 400, { ok: false, error: "Por favor, informe um e-mail válido." });
          return;
        }

        // 1. Procura se o cliente já possui a chave cadastrada no banco para o produto solicitado
        const [allLicenses] = await pool.execute(
          `SELECT * FROM licenses WHERE access_email=? OR email=? ORDER BY expires_at DESC`,
          [email, email]
        );

        const existingProdLicense = allLicenses.find(lic => {
          const keyStr = String(lic.license_key || "").toUpperCase();
          if (requestedProduct === "livecam") {
            return keyStr.startsWith("LIVECAM-");
          } else {
            return keyStr.startsWith("LIVEINF-");
          }
        });

        if (existingProdLicense) {
          json(response, 200, {
            ok: true,
            found: true,
            key: existingProdLicense.license_key,
            plan: existingProdLicense.plan,
            product: requestedProduct === "livecam" ? "Live Cam" : "Live Infinity",
            status: existingProdLicense.status,
            expiresAt: existingProdLicense.expires_at,
            message: `Sua chave do ${requestedProduct === "livecam" ? "Live Cam" : "Live Infinity"} foi encontrada com sucesso!`
          });
          return;
        }

        // 2. Se não encontrou a chave do produto específico, verifica se o cliente possui compra (via licenses ou webhook Cakto ou e-mail de teste)
        let plan = "basic";
        let hasPurchase = false;

        if (email.startsWith("teste") || email.includes("test")) {
          hasPurchase = true;
          plan = "premium";
        } else if (allLicenses.length > 0) {
          hasPurchase = true;
          plan = allLicenses[0].plan || "basic";
        } else {
          const [events] = await pool.execute(
            `SELECT * FROM cakto_webhook_events WHERE customer_email=? AND (payment_status='paid' OR payment_status='approved' OR event_type='purchase_approved' OR event_type='subscription_created') ORDER BY received_at DESC LIMIT 1`,
            [email]
          );
          if (events.length > 0) {
            hasPurchase = true;
            const ev = events[0];
            const payload = typeof ev.payload === "string" ? JSON.parse(ev.payload) : ev.payload;
            plan = ev.mapped_plan || findPlanInPayload(payload) || "basic";
          }
        }

        if (hasPurchase) {
          const id = crypto.randomUUID();
          const key = generateKey(plan, requestedProduct);
          const durationDays = 30;
          const expiresAt = new Date(Date.now() + durationDays * 86400000);
          const amountPaid = plan === "premium" ? 147.00 : plan === "pro" ? 97.00 : 67.00;

          await upsertCustomerAccount(email, plan);
          await pool.execute(
            `INSERT INTO licenses
             (id,account_email,access_email,email,license_key,plan,duration_days,note,amount_paid,net_amount,payment_method,expires_at,status,active)
             VALUES (?,?,?,?,?,?,?,?,?,?,?,?,?,?)`,
            [
              id, email, email, email, key, plan, durationDays,
              `Gerado via autoatendimento (${requestedProduct === "livecam" ? "Live Cam" : "Live Infinity"})`,
              amountPaid, amountPaid * 0.92, "Cakto Self-Service",
              expiresAt, "pending", 1
            ]
          );

          json(response, 200, {
            ok: true,
            created: true,
            key: key,
            plan: plan,
            product: requestedProduct === "livecam" ? "Live Cam" : "Live Infinity",
            expiresAt: expiresAt,
            message: `Sua chave do ${requestedProduct === "livecam" ? "Live Cam" : "Live Infinity"} foi gerada com sucesso!`
          });
          return;
        }

        json(response, 404, {
          ok: false,
          error: "Nenhuma compra ativa foi encontrada para o e-mail informado. Certifique-se de utilizar exatamente o mesmo e-mail preenchido na compra na Cakto."
        });
        return;
      } catch (err) {
        console.error("[claim-license error]", err);
        json(response, 500, { ok: false, error: err.message || String(err), stack: String(err.stack || "") });
        return;
      }
    }

    if (url.pathname === "/api/webhooks/cakto" && request.method === "POST") {
      console.log(`[Webhook Cakto] Nova requisição recebida! IP: ${requestIp(request)}`);
      const body = await readBody(request);
      const suppliedSecret = caktoSecretFromRequest(request, url, body);

      if (
        CAKTO_WEBHOOK_SECRET &&
        suppliedSecret &&
        !safeEqual(suppliedSecret, CAKTO_WEBHOOK_SECRET)
      ) {
        console.warn(`[Webhook Cakto] FALHA DE AUTORIZAÇÃO: Secret incorreto. Fornecido: "${suppliedSecret}", Esperado: "${CAKTO_WEBHOOK_SECRET}"`);
        json(response, 401, {
          ok: false,
          error: "Webhook Cakto não autorizado."
        });
        return;
      }

      const event = caktoEventData(body);
      const plan = findPlanInPayload(body);
      console.log(`[Webhook Cakto] Evento: "${event.eventId}" | Tipo: "${event.eventType}" | Order: "${event.orderId}" | Status: "${event.paymentStatus}" | Email: "${event.email}" | Plano Mapeado: "${plan || 'Não encontrado'}"`);

      try {
        await pool.execute(
          `INSERT INTO cakto_webhook_events (
             event_id,event_type,offer_id,order_id,customer_email,
             payment_status,mapped_plan,payload
           ) VALUES (?,?,?,?,?,?,?,?)
           ON DUPLICATE KEY UPDATE event_id=event_id`,
          [
            event.eventId,
            event.eventType || null,
            event.offerId || null,
            event.orderId || null,
            event.email || null,
            event.paymentStatus || null,
            plan || null,
            JSON.stringify(body)
          ]
        );
      } catch (error) {
        if (error.code === "ER_NO_SUCH_TABLE") {
          json(response, 503, {
            ok: false,
            error: "Execute npm run db:migrate-cakto antes de receber webhooks."
          });
          return;
        }
        throw error;
      }

      if (CAKTO_CAPTURE_ONLY) {
        console.log(`[Webhook Cakto] CAPTURE_ONLY está ATIVO no servidor. Evento salvo no banco de dados, mas nenhuma licença foi gerada/atualizada.`);
        await pool.execute(
          `UPDATE cakto_webhook_events
           SET processed=0,
               processing_error='CAPTURE_ONLY: aguardando validação do payload real'
           WHERE event_id=?`,
          [event.eventId]
        );

        json(response, 200, {
          ok: true,
          received: true,
          captureOnly: true,
          mappedPlan: plan || null,
          message: "Evento salvo. Nenhuma licença foi criada automaticamente."
        });
        return;
      }

      // PROCESS WEBHOOK INLINE (if not capture only)
      let processed = 0;
      let processingError = null;

      try {
        const lowerStatus = String(event.paymentStatus || "").trim().toLowerCase();
        const lowerType = String(event.eventType || "").trim().toLowerCase();

        // Cakto statuses are typically: 'paid', 'approved', 'active', 'completed', 'success' etc.
        const isPaid = lowerStatus === "paid" ||
                       lowerStatus === "approved" ||
                       lowerStatus === "active" ||
                       lowerStatus === "completed" ||
                       lowerStatus === "success" ||
                       lowerType === "charge.paid" ||
                       lowerType === "purchase_approved" ||
                       lowerType === "subscription_created" ||
                       (lowerType === "subscription.status_changed" && lowerStatus === "active");

        const isRevoked = lowerStatus === "refunded" ||
                          lowerStatus === "chargeback" ||
                          lowerStatus === "cancelled" ||
                          lowerStatus === "expired" ||
                          lowerType === "charge.refunded" ||
                          lowerType === "subscription.canceled";

        if (isPaid && event.email && plan) {
          let alreadyProcessed = false;
          if (event.orderId) {
            const [orderCheck] = await pool.execute(
              "SELECT event_id FROM cakto_webhook_events WHERE order_id=? AND processed=1 LIMIT 1",
              [event.orderId]
            );
            if (orderCheck.length > 0) {
              console.log(`[Webhook Cakto] Ignorando evento ${event.eventId} pois o order_id ${event.orderId} já foi processado com sucesso.`);
              processed = 1;
              processingError = `Ignorado: Pedido já processado anteriormente no evento ${orderCheck[0].event_id}`;
              alreadyProcessed = true;
            }
          }

          if (!alreadyProcessed) {
            // 1. Create/update customer account
            const definition = await upsertCustomerAccount(event.email, plan);

            // 2. Check if a license already exists for this email
            const [existing] = await pool.execute(
              `SELECT * FROM licenses WHERE access_email=?`,
              [event.email]
            );

            if (existing.length > 0) {
              // Extend the license
              const licenseRow = existing[0];
              const durationDays = 30; // standard month extension
              const newExpires = new Date(Date.now() + durationDays * 86400000);
              const amountPaid = event.amountPaid || (plan === "premium" ? 147.00 : plan === "pro" ? 97.00 : 67.00);
              const netAmount = event.netAmount || (amountPaid * 0.92);

              await pool.execute(
                `UPDATE licenses
                 SET expires_at=?,
                     active=1,
                     status='active',
                     plan=?,
                     duration_days=?,
                     amount_paid=?,
                     net_amount=?
                 WHERE id=?`,
                [newExpires, plan, durationDays, amountPaid, netAmount, licenseRow.id]
              );

              await upsertCustomerAccount(event.email, plan);
              await pool.execute(
                `UPDATE customer_accounts
                 SET subscription_status='active',
                     current_period_end=?
                 WHERE email=?`,
                [newExpires, event.email]
              );
              console.log(`[Webhook Cakto] Licença existente estendida com SUCESSO para o email ${event.email} (Plano: ${plan}, Valor: R$ ${amountPaid}, Lucro: R$ ${netAmount}, Novo Vencimento: ${newExpires})`);
            } else {
              // Create a new license key
              const id = crypto.randomUUID();
              let key;
              const durationDays = 30;
              const expiresAt = new Date(Date.now() + durationDays * 86400000);
              const amountPaid = event.amountPaid || (plan === "premium" ? 147.00 : plan === "pro" ? 97.00 : 67.00);
              const netAmount = event.netAmount || (amountPaid * 0.92);

              for (let attempt = 0; attempt < 10; attempt += 1) {
                key = generateKey(plan);
                try {
                  await pool.execute(
                    `INSERT INTO licenses
                     (id,account_email,access_email,email,phone,license_key,plan,duration_days,note,amount_paid,net_amount,payment_method,expires_at,status,active)
                     VALUES (?,?,?,?,?,?,?,?,?,?,?,?,?,?,?)`,
                    [
                      id,
                      event.email,
                      event.email,
                      event.email,
                      event.phone || null,
                      key,
                      plan,
                      durationDays,
                      `Criado via webhook Cakto: ${event.orderId || ""}`,
                      amountPaid,
                      netAmount,
                      "Cakto",
                      expiresAt,
                      "pending",
                      1
                    ]
                  );
                  break;
                } catch (error) {
                  if (error.code !== "ER_DUP_ENTRY" || attempt === 9) throw error;
                }
              }

              await upsertCustomerAccount(event.email, plan);
              await pool.execute(
                `UPDATE customer_accounts
                 SET subscription_status='active',
                     current_period_end=?
                 WHERE email=?`,
                [expiresAt, event.email]
              );
              console.log(`[Webhook Cakto] Nova licença criada com SUCESSO para o email ${event.email} (Plano: ${plan}, Key Gerada: ${key}, Valor: R$ ${amountPaid}, Lucro: R$ ${netAmount}, Vencimento: ${expiresAt})`);
            }
            processed = 1;
          }
        } else if (isRevoked && event.email) {
          // Revoga o acesso de TODAS as chaves ligadas a este email (LIVE CAM e LIVE INFINITY)
          await pool.execute(
            `UPDATE licenses
             SET active=0,
                 status='revoked'
             WHERE access_email=? OR account_email=? OR email=?`,
            [event.email, event.email, event.email]
          );

          await pool.execute(
            `UPDATE customer_accounts
             SET subscription_status='cancelled'
             WHERE email=?`,
            [event.email]
          );
          console.log(`[Webhook Cakto] Acesso REVOGADO com SUCESSO para o email ${event.email} (LIVE CAM + LIVE INFINITY bloqueados)`);
          processed = 1;
        } else {
          processingError = `Ignorado: Evento/Status nao mapeados para acao (tipo: ${lowerType}, status: ${lowerStatus}, email: ${event.email || "nulo"}, plano: ${plan || "nulo"})`;
          console.log(`[Webhook Cakto] Evento ignorado (não mapeado): ${processingError}`);
        }
      } catch (e) {
        processed = 0;
        processingError = `Erro no processamento: ${e.message}`;
        console.error(`[Webhook Cakto] EXCEÇÃO ao processar: ${processingError}`);
      }

      // Update event status in database
      await pool.execute(
        `UPDATE cakto_webhook_events
         SET processed=?,
             processing_error=?,
             processed_at=NOW()
         WHERE event_id=?`,
         [processed, processingError, event.eventId]
      );

      json(response, 200, {
        ok: true,
        received: true,
        processed: processed === 1,
        processingError: processingError,
        mappedPlan: plan || null
      });
      return;
    }


    if (url.pathname === "/api/updates/latest" && request.method === "GET") {
      const currentVersion = sanitizeVersion(
        url.searchParams.get("currentVersion")
      ) || "0.0.0";

      const update = await latestPublishedUpdate(request);

      json(response, 200, {
        ok: true,
        currentVersion,
        update,
        updateRequired: Boolean(
          update && compareVersions(update.version, currentVersion) > 0
        ),
        mandatory: Boolean(
          update &&

          if (error.code === "ER_DUP_ENTRY") {
            json(response, 409, {
              ok: false,
              error: "Já existe uma atualização com esta versão."
            });
            return;
          }

          throw error;
        }

        json(response, 201, {
          ok: true,
          update: rowToUpdate({
            id,
            version,
            title: String(body.title || `Live Infinity ${version}`).trim(),
            description: String(body.description || "").trim(),
            changelog: String(body.changelog || "").trim(),
            file_name: fileName,
            file_size: zipBuffer.length,
            mandatory: 1,
            published: body.published === false ? 0 : 1,
            published_at: body.published === false ? null : new Date(),
            created_at: new Date()
          }, request)
        });
        return;
      }

      const adminUpdateMatch = url.pathname.match(
        /^\/api\/admin\/updates\/([^/]+)(?:\/(publish|unpublish))?$/
      );

      if (adminUpdateMatch) {
        const id = decodeURIComponent(adminUpdateMatch[1]);
        const action = adminUpdateMatch[2];

        const [rows] = await pool.execute(
          "SELECT * FROM extension_updates WHERE id=? LIMIT 1",
          [id]
        );

        const update = rows[0];

        if (!update) {
          json(response, 404, {
            ok: false,
            error: "Atualização não encontrada."
          });
          return;
        }

        if (!action && request.method === "DELETE") {
          await pool.execute(
            "DELETE FROM extension_updates WHERE id=?",
            [id]
          );

          const filePath = path.join(UPDATES_DIR, update.file_name);
          if (fs.existsSync(filePath)) fs.unlinkSync(filePath);

          json(response, 200, { ok: true });
          return;
        }

        if (
          (action === "publish" || action === "unpublish") &&
          request.method === "POST"
        ) {
          const published = action === "publish" ? 1 : 0;

          await pool.execute(
            `UPDATE extension_updates
             SET published=?,published_at=?
             WHERE id=?`,
            [published, published ? new Date() : null, id]
          );

          json(response, 200, { ok: true });
          return;
        }
      }

      if (url.pathname === "/api/admin/licenses" && request.method === "GET") {
        const [rows] = await pool.query(
          `SELECT l.*, COALESCE((SELECT SUM(gmv) FROM live_sessions WHERE license_id = l.id), 0.00) AS total_gmv 
           FROM licenses l 
           ORDER BY l.created_at DESC`
        );
        json(response, 200, {
          ok: true,
          licenses: rows.map(rowToLicense)
        });
        return;
      }

      if (url.pathname === "/api/admin/licenses" && request.method === "POST") {
        const body = await readBody(request);
        const accountEmail = normalizeEmail(body.accountEmail || body.email);
        const requestedAccessEmail = normalizeEmail(body.accessEmail);
        const durationDays = Number(body.durationDays);
        const plan = normalizePlan(body.plan);
        const note = String(body.note || "").trim();
        const amountPaid = Math.max(0, Number(body.amountPaid || 0));
        const paymentMethod = String(body.paymentMethod || "").trim();

        if (!accountEmail || !accountEmail.includes("@")) {
          json(response, 400, { ok: false, error: "Digite um e-mail de compra válido." });
          return;
        }
        if (!Number.isInteger(durationDays) || durationDays < 1 || durationDays > 36500) {
          json(response, 400, {
            ok: false,
            error: "Os dias devem estar entre 1 e 36500."
          });
          return;
        }

        const accountPlan = await upsertCustomerAccount(accountEmail, plan);
        const usedKeys = await accountKeyUsage(accountEmail);
        const accessEmail = requestedAccessEmail || suggestedAccessEmail(accountEmail, usedKeys + 1);

        if (!accessEmail || !accessEmail.includes("@")) {
          json(response, 400, { ok: false, error: "Digite um e-mail de acesso válido." });
          return;
        }

        const [existingAccess] = await pool.execute(
          "SELECT id FROM licenses WHERE access_email=? LIMIT 1",
          [accessEmail]
        );

        if (existingAccess.length) {
          json(response, 409, { ok: false, error: "Este e-mail de acesso já está em uso." });
          return;
        }

        if (
          accountPlan.keyLimit !== null &&
          usedKeys >= accountPlan.keyLimit
        ) {
          json(response, 409, {
            ok: false,
            error:
              `O plano ${accountPlan.name} permite ${accountPlan.keyLimit} ` +
              `${accountPlan.keyLimit === 1 ? "chave" : "chaves"} por conta.`
          });
          return;
        }

        const id = crypto.randomUUID();
        let key;

        for (let attempt = 0; attempt < 10; attempt += 1) {
          key = generateKey(plan);
          try {
            await pool.execute(
              `INSERT INTO licenses
               (id,account_email,access_email,email,license_key,plan,duration_days,note,amount_paid,net_amount,payment_method)
               VALUES (?,?,?,?,?,?,?,?,?,?,?)`,
              [id,accountEmail,accessEmail,accessEmail,key,plan,durationDays,note || null,amountPaid,amountPaid,paymentMethod || null]
            );
            break;
          } catch (error) {
            if (error.code !== "ER_DUP_ENTRY" || attempt === 9) throw error;
          }
        }

        const row = await findLicenseByKey(key);
        await logEvent(request, "license_created", id, accessEmail, { accountEmail, accessEmail, plan, durationDays });
        json(response, 201, { ok: true, license: rowToLicense(row) });
        return;
      }

      if (url.pathname === "/api/admin/support" && request.method === "GET") {
        const [rows] = await pool.query(
          "SELECT * FROM support_tickets ORDER BY created_at DESC"
        );
        json(response, 200, {
          ok: true,
          tickets: rows.map(row => ({
            id: row.id,
            subject: row.subject,
            customerEmail: row.customer_email,
            message: row.message,
            priority: row.priority,
            status: row.status,
            createdAt: toIso(row.created_at),
            updatedAt: toIso(row.updated_at)
          }))
        });
        return;
      }

      if (url.pathname === "/api/admin/support" && request.method === "POST") {
        const body = await readBody(request);
        const subject = String(body.subject || "").trim();
        const customerEmail = normalizeEmail(body.customerEmail);
        const message = String(body.message || "").trim();
        const priority = ["low","normal","high"].includes(body.priority)
          ? body.priority
          : "normal";

        if (!subject || !message) {
          json(response, 400, {
            ok: false,
            error: "Assunto e mensagem são obrigatórios."
          });
          return;
        }

        const id = crypto.randomUUID();
        await pool.execute(
          `INSERT INTO support_tickets
           (id,subject,customer_email,message,priority)
           VALUES (?,?,?,?,?)`,
          [id,subject,customerEmail || null,message,priority]
        );

        json(response, 201, { ok: true, ticket: { id } });
        return;
      }
      const supportMatch = url.pathname.match(/^\/api\/admin\/support\/([^/]+)$/);

      if (supportMatch && request.method === "PATCH") {
        const body = await readBody(request);
        const status = ["open","in_progress","resolved"].includes(body.status)
          ? body.status
          : null;
        const priority = ["low","normal","high"].includes(body.priority)
          ? body.priority
          : null;

        if (!status && !priority) {
          json(response, 400, { ok: false, error: "Nada para atualizar." });
          return;
        }

        await pool.execute(
          `UPDATE support_tickets
           SET status=COALESCE(?,status),priority=COALESCE(?,priority)
           WHERE id=?`,
          [status,priority,decodeURIComponent(supportMatch[1])]
        );

        json(response, 200, { ok: true });
        return;
      }

      const adminMsgMatch = url.pathname.match(/^\/api\/admin\/support\/([^/]+)\/messages$/);
      if (adminMsgMatch) {
        const ticketId = decodeURIComponent(adminMsgMatch[1]);

        if (request.method === "GET") {
          const [messages] = await pool.execute(
            "SELECT * FROM support_messages WHERE ticket_id=? ORDER BY created_at ASC",
            [ticketId]
          );

          json(response, 200, {
            ok: true,
            messages: messages.map(m => ({
              id: m.id,
              sender: m.sender,
              message: m.message,
              createdAt: toIso(m.created_at)
            }))
          });
          return;
        }

        if (request.method === "POST") {
          const body = await readBody(request);
          const message = String(body.message || "").trim();

          if (!message) {
            json(response, 400, { ok: false, error: "Mensagem é obrigatória." });
            return;
          }

          const msgId = crypto.randomUUID();
          await pool.execute(
            "INSERT INTO support_messages (id, ticket_id, sender, message) VALUES (?, ?, 'agent', ?)",
            [msgId, ticketId, message]
          );

          json(response, 201, { ok: true });
          return;
        }
      }

            const releaseDeviceMatch = url.pathname.match(
        /^\/api\/admin\/licenses\/([^/]+)\/release-device$/
      );

      if (releaseDeviceMatch && request.method === "POST") {
        const id = releaseDeviceMatch[1];

        const [rows] = await pool.execute(
          "SELECT * FROM licenses WHERE id=? LIMIT 1",
          [id]
        );

        if (!rows.length) {
          json(response, 404, {
            ok: false,
            error: "Licença não encontrada."
          });
          return;
        }

        await pool.execute(
          `UPDATE licenses
           SET device_id=NULL,device_name=NULL,device_fingerprint=NULL,last_validation_at=NULL
           WHERE id=?`,
          [id]
        );

        await logEvent(
          request,
          "device_released",
          id,
          rows[0].access_email || rows[0].email
        );

        json(response, 200, { ok: true });
        return;
      }

const licenseMatch = url.pathname.match(
        /^\/api\/admin\/licenses\/([^/]+)(?:\/(renew|revoke|restore|reset-device))?$/
      );

      if (licenseMatch) {
        const id = decodeURIComponent(licenseMatch[1]);
        const action = licenseMatch[2];

        const [rows] = await pool.execute(
          "SELECT * FROM licenses WHERE id=? LIMIT 1",
          [id]
        );
        const row = rows[0];

        if (!row) {
          json(response, 404, { ok: false, error: "Licença não encontrada." });
          return;
        }

        if (!action && request.method === "DELETE") {
          await pool.execute("DELETE FROM licenses WHERE id=?", [id]);
          await logEvent(request, "license_deleted", id, row.email);
          json(response, 200, { ok: true });
          return;
        }

        if (!action && request.method === "PATCH") {
          const body = await readBody(request);
          const plan =
            body.plan !== undefined
              ? normalizePlan(body.plan)
              : null;

          let newExpires = null;
          let durationDays = null;
          if (body.durationDays !== undefined) {
            durationDays = Number(body.durationDays);
            if (Number.isInteger(durationDays) && durationDays >= 0) {
              const baseDate = new Date(row.created_at || Date.now());
              newExpires = new Date(baseDate.getTime() + durationDays * 86400000);
            } else {
              durationDays = null;
            }
          }

          await pool.execute(
            `UPDATE licenses SET
               plan=COALESCE(?,plan),
               note=COALESCE(?,note),
               phone=COALESCE(?,phone),
               amount_paid=COALESCE(?,amount_paid),
               payment_method=COALESCE(?,payment_method),
               duration_days=COALESCE(?,duration_days),
               expires_at=COALESCE(?,expires_at)
             WHERE id=?`,
            [
              plan,
              body.note !== undefined ? String(body.note || "") : null,
              body.phone !== undefined ? String(body.phone || "").replace(/\D/g, "") : null,
              body.amountPaid !== undefined ? Math.max(0, Number(body.amountPaid || 0)) : null,
              body.paymentMethod !== undefined ? String(body.paymentMethod || "") : null,
              durationDays,
              newExpires,
              id
            ]
          );

          if (plan) {
            const ownerEmail = row.account_email || row.email;
            await upsertCustomerAccount(ownerEmail, plan);
            await pool.execute(
              "UPDATE licenses SET plan=? WHERE account_email=?",
              [plan, ownerEmail]
            );
          }

          const [updatedRows] = await pool.execute(
            "SELECT * FROM licenses WHERE id=? LIMIT 1",
            [id]
          );
          json(response, 200, {
            ok: true,
            license: rowToLicense(updatedRows[0])
          });
          return;
        }

        if (action === "renew" && request.method === "POST") {
          const body = await readBody(request);
          const days = Number(body.days);

          if (!Number.isInteger(days) || days < 1 || days > 36500) {
            json(response, 400, { ok: false, error: "Quantidade de dias inválida." });
            return;
          }

          if (!row.activated_at) {
            await pool.execute(
              "UPDATE licenses SET duration_days=duration_days+?,active=1,status='pending' WHERE id=?",
              [days,id]
            );
          } else {
            const base = row.expires_at && new Date(row.expires_at) > new Date()
              ? new Date(row.expires_at)
              : new Date();
            base.setDate(base.getDate() + days);

            await pool.execute(
              "UPDATE licenses SET expires_at=?,active=1,status='active' WHERE id=?",
              [base,id]
            );
          }

          json(response, 200, { ok: true });
          return;
        }

        if (action === "revoke" && request.method === "POST") {
          await pool.execute(
            "UPDATE licenses SET active=0,status='revoked' WHERE id=?",
            [id]
          );
          json(response, 200, { ok: true });
          return;
        }

        if (action === "restore" && request.method === "POST") {
          const status = row.activated_at ? "active" : "pending";
          await pool.execute(
            "UPDATE licenses SET active=1,status=? WHERE id=?",
            [status,id]
          );
          json(response, 200, { ok: true });
          return;
        }

        if (action === "reset-device" && request.method === "POST") {
          await pool.execute(
            "UPDATE licenses SET device_id=NULL,device_fingerprint=NULL WHERE id=?",
            [id]
          );
          json(response, 200, { ok: true });
          return;
        }
      }

      json(response, 404, {
        ok: false,
        error: "Rota administrativa não encontrada."
      });
      return;
    }

    serveAdminStatic(url.pathname, response);
  } catch (error) {
    console.error(error);
    json(response, 500, {
      ok: false,
      error: "Erro interno do servidor."
    });
  }
});
